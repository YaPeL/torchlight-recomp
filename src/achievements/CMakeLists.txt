# Supply a separately licensed Valve SDK. Nothing is loaded from reference/pc.
set(STEAMWORKS_SDK_ROOT "" CACHE PATH "External Steamworks SDK root")
option(TORCHLIGHT_STEAM_WRITES "Compile native Steam mutation calls (runtime gate also required)" OFF)
target_sources(torchlight_achievements PRIVATE steam_backend.cpp steam_native.cpp)
if(STEAMWORKS_SDK_ROOT)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Native Steam adapter discovery and durable journal currently support Linux x86_64")
  endif()
  if(NOT EXISTS "${STEAMWORKS_SDK_ROOT}/public/steam/steam_api.h")
    message(FATAL_ERROR "STEAMWORKS_SDK_ROOT must contain public/steam/steam_api.h")
  endif()
  file(REAL_PATH "${STEAMWORKS_SDK_ROOT}" _achievement_steam_sdk)
  if(_achievement_steam_sdk MATCHES "/reference(/|$)")
    message(FATAL_ERROR "The PC reference cannot supply the Steam implementation")
  endif()
  target_include_directories(torchlight_achievements SYSTEM PRIVATE "${STEAMWORKS_SDK_ROOT}/public")
  find_library(_achievement_steam_library steam_api PATHS
    "${STEAMWORKS_SDK_ROOT}/redistributable_bin/linux64" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(torchlight_achievements PUBLIC "${_achievement_steam_library}")
  target_compile_definitions(torchlight_achievements PRIVATE TORCHLIGHT_HAS_STEAMWORKS=1)
  if(TORCHLIGHT_STEAM_WRITES)
    target_compile_definitions(torchlight_achievements PRIVATE TORCHLIGHT_STEAM_WRITES=1)
  endif()
elseif(TORCHLIGHT_STEAM_WRITES)
  message(FATAL_ERROR "Native Steam writes require an external Steamworks SDK")
endif()
add_executable(achievement_steam_test steam_backend_test.cpp)
target_link_libraries(achievement_steam_test PRIVATE torchlight_achievements)
target_compile_options(achievement_steam_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME achievement_steam_test COMMAND achievement_steam_test)
